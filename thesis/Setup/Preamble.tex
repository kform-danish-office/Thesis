% Thesis drafting helpers. Set \showdraftnotestrue while writing.
% The default clean build keeps evidence gaps visible in the source, but removes
% draft-note boxes and heavy placeholder figures from the PDF.
\newif\ifshowdraftnotes
\showdraftnotesfalse
\newcommand{\draftnote}[1]{%
  \ifshowdraftnotes
  \par\smallskip\noindent
  \fcolorbox{orange}{orange!10}{%
    \parbox{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}{%
      \textbf{Draft note:} #1}}
  \par\smallskip
  \fi}
\newcommand{\placeholdervalue}[1]{%
  \ifshowdraftnotes
    \textcolor{red}{\textbf{[#1]}}%
  \else
    \textcolor{black!45}{\emph{[#1]}}%
  \fi}
\newcommand{\placeholderfigure}[2][0.20\textheight]{%
  \ifshowdraftnotes
  \begin{figure}[H]
  \centering
  \fbox{\parbox[c][#1][c]{0.90\linewidth}{\centering\textbf{Figure placeholder}\\[4pt]#2}}
  \caption{#2}
  \end{figure}
  \fi}
