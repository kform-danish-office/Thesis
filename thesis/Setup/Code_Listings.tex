% listings measures a complete physical source line even when it wraps.
% Bound that bookkeeping width for wrapped listings to avoid TeX dimension
% overflow on long HTML lines; the original input and line numbers are retained.
% Activate the Unicode parallel symbol so the listings substitution is used.
\makeatletter
\lst@AddToHookAtTop{SelectCharTable}{\catcode`∥=\active}
% Allow a wrapped physical source line to continue onto the next page.
\lst@AddToHook{Init}{\lst@ifbreaklines\brokenpenalty=0\relax\fi}
\lst@AddToHook{PostOutput}{%
  \lst@ifbreaklines
    \ifdim\lst@currlwidth>\linewidth
      \global\lst@currlwidth\linewidth
    \fi
  \fi}
\makeatother
